# Decider — DFA visual review request

A self-contained single-file DFA checker (HTML + CSS + JS + `dfa.py`, no
dependencies, no build step, no network). I want feedback on the **visual
design**: layout, hierarchy, typography, spacing, colour, the state diagram, and
dark/light theming.

**The machine** — accepts every binary string that ends in `1`. Two states only,
no dead state:

```
Sigma = {0, 1}   Q = {q0, q1}   start q0   accepting F = {q1}

        input 0   input 1
  -> q0    q0       q1
  * q1    q0       q1
```

Acceptance is decided from the **final state**, never from the last character.

**What I would like**
1. Where the visual hierarchy is weak or ambiguous.
2. Whether the state diagram reads clearly — spacing, arrow labels, the double
   ring, the start arrow.
3. Colour and contrast in both themes; anything that fails WCAG.
4. Concrete CSS changes, not just opinions.

The machine recognises every binary string that ends in 1.

    delta(q0, 0) = q0     delta(q0, 1) = q1
    delta(q1, 0) = q0     delta(q1, 1) = q1

    q0  the input read so far is empty or ends in 0
    q1  the input read so far ends in 1

A string is accepted exactly when the state it finishes in is q1, which is the
same thing as saying the string ends in 1.

There is no dead state, and that is a property of this language rather than an
omission. An earlier build of this file recognised the single exact string "101"
and needed a trap, because any extra symbol had to be a failure; and a later
build recognised "contains 101" with an absorbing q3. Neither is this machine.
Here every state is reachable on both symbols and no input can ever be stuck, so
a trap would only reject perfectly good strings.

This module is the single source of truth. The application's figure is generated
from DFA["delta"] by script.js, and nothing in JavaScript decides anything.

from typing import Any, Dict, List

# DFA definition, exactly as the assignment states it.
DFA = {
    "states": ["q0", "q1"],
    "alphabet": ["0", "1"],
    "start": "q0",
    "accepting": ["q1"],
    "delta": {
        "q0": {"0": "q0", "1": "q1"},
        "q1": {"0": "q0", "1": "q1"},
    },
}

ACCEPTED_DESCRIPTION = "This machine accepts every binary string that ends in 1."


def validate(input_str: str) -> Dict[str, Any]:
    """
    Validate that every character of input_str is in the alphabet {0, 1}.

    Args:
        input_str: the string to validate, unmodified (no trimming).

    Returns:
        {'valid': bool, 'invalid': [{'symbol', 'index'}, ...]} with 1-based
        positions. A space is reported as symbol ' ' and the interface renders
        that as "space".
    """
    invalid = [
        {"symbol": char, "index": i}
        for i, char in enumerate(input_str, start=1)
        if char not in DFA["alphabet"]
    ]
    return {"valid": not invalid, "invalid": invalid}


def simulate(input_str: str) -> Dict[str, Any]:
    """
    Simulate the machine over a valid input string, one symbol at a time.

    Args:
        input_str: a string containing only '0' and '1'.

    Returns:
        {'input', 'startState', 'steps', 'flow', 'finalState', 'accepted',
         'description'}.
        steps is [{'step', 'from', 'symbol', 'to'}] numbered from 1; flow has
        one more entry than steps, starting at the start state.
        Empty input yields steps [], flow ['q0'], finalState 'q0', and
        accepted False, because the empty string ends in nothing and is not a
        member of the language.
    """
    start_state = DFA["start"]

    def result(steps: List[Dict[str, Any]], flow: List[str], final_state: str) -> Dict[str, Any]:
        return {
            "input": input_str,
            "startState": start_state,
            "steps": steps,
            "flow": flow,
            "finalState": final_state,
            "accepted": final_state in DFA["accepting"],
            "description": ACCEPTED_DESCRIPTION,
        }

    # Empty input: no transitions at all, so the machine is still at the start.
    if not input_str:
        return result([], [start_state], start_state)

    steps: List[Dict[str, Any]] = []
    flow = [start_state]
    current = start_state

    for i, symbol in enumerate(input_str, start=1):
        nxt = DFA["delta"][current][symbol]
        steps.append({"step": i, "from": current, "symbol": symbol, "to": nxt})
        current = nxt
        flow.append(current)

    return result(steps, flow, current)


def check(input_str: str) -> Dict[str, Any]:
    """
    Validate, then simulate. This is what the bridge calls.

    Args:
        input_str: the string to check, unmodified.

    Returns:
        {'status': 'invalid', 'errors': [{'symbol', 'index'}, ...]} when any
        symbol is outside the alphabet - an invalid string is never reported as
        accepted or rejected, because the machine never saw it.
        Otherwise {'status': 'valid', 'input', 'start', 'steps', 'flow',
        'final', 'accepted', 'description'}.
    """
    validation = validate(input_str)
    if not validation["valid"]:
        return {"status": "invalid", "errors": validation["invalid"]}

    sim = simulate(input_str)
    return {
        "status": "valid",
        "input": sim["input"],
        "start": sim["startState"],
        "steps": sim["steps"],
        "flow": sim["flow"],
        "final": sim["finalState"],
        "accepted": sim["accepted"],
        "description": sim["description"],
    }

/* The machine. This is the same table dfa.py defines in DFA["delta"]. */
const DFA = {
  alphabet:  ["0", "1"],
  start:     "q0",
  accepting: ["q1"],
  delta: {
    q0: { "0": "q0", "1": "q1" },
    q1: { "0": "q0", "1": "q1" },

const ACCEPTED_DESCRIPTION = "This machine accepts every binary string that ends in 1.";

/* ---- dfa.py: validate() -------------------------------------------------- */
function validate(str) {
  const invalid = [];
  for (let i = 0; i < str.length; i++) {
    if (!DFA.alphabet.includes(str[i])) invalid.push({ symbol: str[i], index: i + 1 });
  }
  return { valid: invalid.length === 0, invalid: invalid };
}

/* ---- dfa.py: simulate() -------------------------------------------------- */
function simulate(str) {
  const startState = DFA.start;
  const finish = (steps, flow, finalState) => ({
    input: str,
    startState: startState,
    steps: steps,
    flow: flow,
    finalState: finalState,
    accepted: DFA.accepting.indexOf(finalState) !== -1,
    description: ACCEPTED_DESCRIPTION,
  });

  // Empty input: no transitions at all, so the machine is still at the start.
  if (!str) return finish([], [startState], startState);

  const steps = [];
  const flow = [startState];
  let current = startState;
  for (let i = 0; i < str.length; i++) {
    const symbol = str[i];
    const next = DFA.delta[current][symbol];
    steps.push({ step: i + 1, from: current, symbol: symbol, to: next });
    current = next;
    flow.push(current);
  }
  return finish(steps, flow, current);
}

/* ---- dfa.py: check() ----------------------------------------------------- */
function check(str) {
  const validation = validate(str);
  // An invalid string is never reported as accepted or rejected: the machine
  // never saw it, because validation runs before the automaton.
  if (!validation.valid) return { status: "invalid", errors: validation.invalid };

  const sim = simulate(str);
  return {
    status: "valid",
    input: sim.input,
    start: sim.startState,
    steps: sim.steps,
    flow: sim.flow,
    final: sim.finalState,
    accepted: sim.accepted,
    description: sim.description,
  };
}

function selfLoop(c) {

  const start = { x: c.x - dx, y: c.y - dy };
  const end = { x: c.x + dx, y: c.y - dy };
  const apex = (2 * start.y + 6 * top) / 8;
  return { d: `M${round2(start.x)},${round2(start.y)} C${round2(c.x - reach)},${round2(top)} ` +
             `${round2(c.x + reach)},${round2(top)} ${round2(end.x)},${round2(end.y)}`,
           label: { x: c.x, y: round2(apex - 17) } };
}

/* The straight run between two circles on the same row. */
function straight(from, to) {
  const x0 = from.x + from.outer;
  const x1 = to.x - to.outer;
  return { d: `M${round2(x0)},${round2(from.y)} L${round2(x1)},${round2(to.y)}`,
           label: { x: (x0 + x1) / 2, y: from.y - 13 } };
}

/* The way back: one arc below both circles, symbol under the arc. Each end
   leaves the circle on the side facing the other, on its outer edge. */
const BACK_ANGLE = (125 * Math.PI) / 180;
const BACK_DIP = 60;
function backCurve(from, to) {
  const a = BACK_ANGLE;
  const b = Math.PI - BACK_ANGLE;
  const start = { x: from.x + from.outer * Math.cos(a), y: from.y + from.outer * Math.sin(a) };
  const end = { x: to.x + to.outer * Math.cos(b), y: to.y + to.outer * Math.sin(b) };
  const bottom = (start.y + 3 * (start.y + BACK_DIP) + 3 * (end.y + BACK_DIP) + end.y) / 8;
  return { d: `M${round2(start.x)},${round2(start.y)} C${round2(start.x)},${round2(start.y + BACK_DIP)} ` +
             `${round2(end.x)},${round2(end.y + BACK_DIP)} ${round2(end.x)},${round2(end.y)}`,
           label: { x: (start.x + end.x) / 2, y: round2(bottom + 24) } };
}

/* Read Table 1, so the drawing cannot disagree with the machine shown beside
   it. Change a cell in the table and the figure changes with it. */
function readTable() {
  const head = [...document.querySelectorAll("#trans-table thead th")]
    .map((th, i) => (i === 0 ? null : th.textContent.replace(/^Input\s*/i, "").trim()))
    .filter(Boolean);
  return [...document.querySelectorAll("#trans-table tbody tr")].map((tr) => {
    const cells = [...tr.children];
    const state = (cells[0].textContent.match(/q[0-9a-z]+/i) || [])[0];
    const inputs = {};
    head.forEach((symbol, i) => { inputs[symbol] = (cells[i + 1].textContent.match(/q[0-9a-z]+/i) || [])[0]; });
    return { state: state, inputs: inputs };
  });
}

function shapeFor(from, to) {
  const a = circleOf("node-" + from);
  const b = circleOf("node-" + to);
  if (from === to) return selfLoop(a);
  return Math.abs(a.y - b.y) < 1 ? (b.x > a.x ? straight(a, b) : backCurve(a, b)) : straight(a, b);
}

/* Generate every arrow and its label, once. One path per transition, with a
   stable id, so a re-render can never double one. */
function buildFigure() {
  const layer = document.getElementById("edge-layer");
  layer.textContent = "";
  EDGES.length = 0;
  edgeIds = {};

  const rows = readTable();
  const first = rows[0].state;

  const q0 = circleOf("node-" + first);
  const lead = q0.x - q0.outer - 46;
  const start = svgEl("path", { id: "edge-start", class: "edge", pathLength: 1,
    d: `M${round2(lead)},${round2(q0.y)} L${round2(q0.x - q0.outer)},${round2(q0.y)}`,
    "marker-end": "url(#arrow)" });
  layer.append(start);

  const entries = [];
  rows.forEach((row) => {
    Object.keys(row.inputs).forEach((symbol) => {
      const dest = row.inputs[symbol];
      entries.push({ id: "edge-" + row.state + "-" + dest + "-" + symbol,
                     from: row.state, to: dest, symbol: symbol });
    });
  });
  // Draw the top row first, then everything else, then the self-loops.
  entries.sort((x, y) => (x.from === x.to ? 2 : 0) - (y.from === y.to ? 2 : 0));

  entries.forEach((entry) => {
    const shape = shapeFor(entry.from, entry.to);
    const path = svgEl("path", { id: entry.id, class: "edge", d: shape.d,
                                 pathLength: 1, "marker-end": "url(#arrow)" });
    layer.append(path);
    const label = svgEl("text", { id: "label-" + entry.id, class: "edge-label",
      x: round2(shape.label.x), y: round2(shape.label.y), "text-anchor": "middle" });
    label.textContent = entry.symbol;
    layer.append(label);
    EDGES.push({ id: entry.id, from: entry.from, to: entry.to, path: path, label: label });
    edgeIds[entry.from + "-" + entry.to + "-" + entry.symbol] = path;
  });
}

function parkOn(id) {
  const path = document.getElementById(id);
  if (!path) return;
  const end = path.getPointAtLength(path.getTotalLength());
  el.token.style.transform = `translate(${end.x}px, ${end.y}px)`;

function runWalk(steps) {
  clearTimers();
  EDGES.forEach((e) => e.path.classList.remove("edge--taken"));
  hideToken();
  const rows = [...el.steps.querySelectorAll("tr.step")];

  const settle = () => {
    clearTimers();
    rows.forEach((r, i) => {
      r.classList.remove("step--pending", "step--current");
      if (i > 0) r.classList.add("step--done");
    });
    if (steps.length) {
      const last = steps[steps.length - 1];
      const id = "edge-" + last.from + "-" + last.to + "-" + last.symbol;
      setTakenEdge(id);
      parkOn(id);
    }
  };

  if (!steps.length) { settle(); return; }

  steps.forEach((step, i) => {
    timers.push(setTimeout(() => {
      rows.forEach((r, k) => {
        r.classList.remove("step--current");
        if (k > 0 && k <= i) r.classList.add("step--done");
      });
      rows[i + 1].classList.add("step--current");
      const id = "edge-" + step.from + "-" + step.to + "-" + step.symbol;
      setTakenEdge(id);
      const path = document.getElementById(id);
      if (path) {
        const end = path.getPointAtLength(path.getTotalLength());
        el.token.style.transform = `translate(${end.x}px, ${end.y}px)`;
        el.token.classList.add("is-moving");
      }
    }, i * STEP_MS));
  });
  timers.push(setTimeout(settle, steps.length * STEP_MS + 40));
}

function copyPython() {
  const text = pythonSource();
  const done = (ok) => {
    if (ok) { el.copyNote.textContent = "dfa.py copied to the clipboard."; return; }
    // Last resort: select the source itself so Ctrl+C works straight away.
    selectSource();
    el.copyNote.textContent = "The browser blocked the clipboard - the source is now selected, press Ctrl+C (Cmd+C on a Mac).";
  };
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(text).then(() => done(true), () => done(fallbackCopy(text)));
  } else {
    done(fallbackCopy(text));
  }
}

// execCommand is the path that still works when the page is opened straight from
// disk, where the async clipboard API is blocked as an insecure context. The
// textarea must NOT be readonly: a readonly textarea is refused by execCommand in
// some engines, which is exactly the failure this replaced.
function fallbackCopy(text) {
  const ta = document.createElement("textarea");
  ta.value = text;
  ta.style.position = "fixed";
  ta.style.top = "-1000px";
  ta.style.opacity = "0";
  document.body.append(ta);
  ta.select();
  ta.setSelectionRange(0, text.length);
  let ok = false;
  try { ok = document.execCommand("copy"); } catch (e) { ok = false; }
  document.body.removeChild(ta);
  return ok;
}

function selectSource() {
  const pre = document.getElementById("dfa-py");
  const range = document.createRange();
  range.selectNodeContents(pre);
  const sel = window.getSelection();
  sel.removeAllRanges();
  sel.addRange(range);
}

function downloadPython() {
  const blob = new Blob([pythonSource() + "\n"], { type: "text/x-python;charset=utf-8" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = "dfa.py";
  document.body.append(a);
  a.click();
  document.body.removeChild(a);
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

  alphabet:  ["0", "1"],
  start:     "q0",
  accepting: ["q1"],
  delta: {
    q0: { "0": "q0", "1": "q1" },
    q1: { "0": "q0", "1": "q1" },
  },
};

const ACCEPTED_DESCRIPTION = "This machine accepts every binary string that ends in 1.";

/* ---- dfa.py: validate() -------------------------------------------------- */
function validate(str) {
  const invalid = [];
  for (let i = 0; i < str.length; i++) {
    if (!DFA.alphabet.includes(str[i])) invalid.push({ symbol: str[i], index: i + 1 });
  }
  return { valid: invalid.length === 0, invalid: invalid };
}

/* ---- dfa.py: simulate() -------------------------------------------------- */
function simulate(str) {
  const startState = DFA.start;
  const finish = (steps, flow, finalState) => ({
    input: str,
    startState: startState,
    steps: steps,
    flow: flow,
    finalState: finalState,
    accepted: DFA.accepting.indexOf(finalState) !== -1,
    description: ACCEPTED_DESCRIPTION,
  });

  // Empty input: no transitions at all, so the machine is still at the start.
  if (!str) return finish([], [startState], startState);

  const steps = [];
  const flow = [startState];
  let current = startState;
  for (let i = 0; i < str.length; i++) {
    const symbol = str[i];
    const next = DFA.delta[current][symbol];
    steps.push({ step: i + 1, from: current, symbol: symbol, to: next });
    current = next;
    flow.push(current);
  }
  return finish(steps, flow, current);
}

/* ---- dfa.py: check() ----------------------------------------------------- */
function check(str) {
  const validation = validate(str);
  // An invalid string is never reported as accepted or rejected: the machine
  // never saw it, because validation runs before the automaton.
  if (!validation.valid) return { status: "invalid", errors: validation.invalid };

  const sim = simulate(str);
  return {
    status: "valid",
    input: sim.input,
    start: sim.startState,
    steps: sim.steps,
    flow: sim.flow,
    final: sim.finalState,
    accepted: sim.accepted,
    description: sim.description,
  };
}

function circleOf(id) {

/* One self-loop, above its circle, anchored clear of the ring, with its symbol
   above the loop. */
function selfLoop(c) {
  const anchor = c.outer + 5;
  const dx = anchor * Math.sin(Math.PI / 3);
  const dy = anchor * Math.cos(Math.PI / 3);
  const top = c.y - (anchor + 44);
  const reach = dx * 1.25;
  const start = { x: c.x - dx, y: c.y - dy };
  const end = { x: c.x + dx, y: c.y - dy };
  const apex = (2 * start.y + 6 * top) / 8;
  return { d: `M${round2(start.x)},${round2(start.y)} C${round2(c.x - reach)},${round2(top)} ` +
             `${round2(c.x + reach)},${round2(top)} ${round2(end.x)},${round2(end.y)}`,
           label: { x: c.x, y: round2(apex - 17) } };
}

/* The straight run between two circles on the same row. */
function straight(from, to) {
  const x0 = from.x + from.outer;
  const x1 = to.x - to.outer;
  return { d: `M${round2(x0)},${round2(from.y)} L${round2(x1)},${round2(to.y)}`,
           label: { x: (x0 + x1) / 2, y: from.y - 13 } };
}

/* The way back: one arc below both circles, symbol under the arc. Each end
   leaves the circle on the side facing the other, on its outer edge. */
const BACK_ANGLE = (125 * Math.PI) / 180;
const BACK_DIP = 60;
function backCurve(from, to) {
  const a = BACK_ANGLE;
  const b = Math.PI - BACK_ANGLE;
  const start = { x: from.x + from.outer * Math.cos(a), y: from.y + from.outer * Math.sin(a) };
  const end = { x: to.x + to.outer * Math.cos(b), y: to.y + to.outer * Math.sin(b) };
  const bottom = (start.y + 3 * (start.y + BACK_DIP) + 3 * (end.y + BACK_DIP) + end.y) / 8;
  return { d: `M${round2(start.x)},${round2(start.y)} C${round2(start.x)},${round2(start.y + BACK_DIP)} ` +
             `${round2(end.x)},${round2(end.y + BACK_DIP)} ${round2(end.x)},${round2(end.y)}`,
           label: { x: (start.x + end.x) / 2, y: round2(bottom + 24) } };
}

/* Read Table 1, so the drawing cannot disagree with the machine shown beside
   it. Change a cell in the table and the figure changes with it. */
function readTable() {
  const head = [...document.querySelectorAll("#trans-table thead th")]
    .map((th, i) => (i === 0 ? null : th.textContent.replace(/^Input\s*/i, "").trim()))
    .filter(Boolean);
  return [...document.querySelectorAll("#trans-table tbody tr")].map((tr) => {
    const cells = [...tr.children];
    const state = (cells[0].textContent.match(/q[0-9a-z]+/i) || [])[0];
    const inputs = {};
    head.forEach((symbol, i) => { inputs[symbol] = (cells[i + 1].textContent.match(/q[0-9a-z]+/i) || [])[0]; });
    return { state: state, inputs: inputs };
  });
}

function shapeFor(from, to) {
  const a = circleOf("node-" + from);
  const b = circleOf("node-" + to);
  if (from === to) return selfLoop(a);
  return Math.abs(a.y - b.y) < 1 ? (b.x > a.x ? straight(a, b) : backCurve(a, b)) : straight(a, b);
}

  layer.textContent = "";
  EDGES.length = 0;
  edgeIds = {};

  const rows = readTable();
  const first = rows[0].state;

  const q0 = circleOf("node-" + first);
  const lead = q0.x - q0.outer - 46;
  const start = svgEl("path", { id: "edge-start", class: "edge", pathLength: 1,
    d: `M${round2(lead)},${round2(q0.y)} L${round2(q0.x - q0.outer)},${round2(q0.y)}`,
    "marker-end": "url(#arrow)" });
  layer.append(start);

  const entries = [];
  rows.forEach((row) => {
    Object.keys(row.inputs).forEach((symbol) => {
      const dest = row.inputs[symbol];
      entries.push({ id: "edge-" + row.state + "-" + dest + "-" + symbol,
                     from: row.state, to: dest, symbol: symbol });
    });
  });
  // Draw the top row first, then everything else, then the self-loops.
  entries.sort((x, y) => (x.from === x.to ? 2 : 0) - (y.from === y.to ? 2 : 0));

  entries.forEach((entry) => {
    const shape = shapeFor(entry.from, entry.to);
    const path = svgEl("path", { id: entry.id, class: "edge", d: shape.d,
                                 pathLength: 1, "marker-end": "url(#arrow)" });

    label.textContent = entry.symbol;
    layer.append(label);
    EDGES.push({ id: entry.id, from: entry.from, to: entry.to, path: path, label: label });
    edgeIds[entry.from + "-" + entry.to + "-" + entry.symbol] = path;
  });
}

"""DFA logic for Decider (AL102 automata theory midterm).

The machine recognises every binary string that ends in 1.

    delta(q0, 0) = q0     delta(q0, 1) = q1
    delta(q1, 0) = q0     delta(q1, 1) = q1

    q0  the input read so far is empty or ends in 0
    q1  the input read so far ends in 1

A string is accepted exactly when the state it finishes in is q1, which is the
same thing as saying the string ends in 1.

There is no dead state, and that is a property of this language rather than an
omission. An earlier build of this file recognised the single exact string "101"
and needed a trap, because any extra symbol had to be a failure; and a later
build recognised "contains 101" with an absorbing q3. Neither is this machine.
Here every state is reachable on both symbols and no input can ever be stuck, so
a trap would only reject perfectly good strings.

This module is the single source of truth. The application's figure is generated
from DFA["delta"] by script.js, and nothing in JavaScript decides anything.
"""

from typing import Any, Dict, List

# DFA definition, exactly as the assignment states it.
DFA = {
    "states": ["q0", "q1"],
    "alphabet": ["0", "1"],
    "start": "q0",
    "accepting": ["q1"],
    "delta": {
        "q0": {"0": "q0", "1": "q1"},
        "q1": {"0": "q0", "1": "q1"},
    },
}

ACCEPTED_DESCRIPTION = "This machine accepts every binary string that ends in 1."


def validate(input_str: str) -> Dict[str, Any]:
    """
    Validate that every character of input_str is in the alphabet {0, 1}.

    Args:
        input_str: the string to validate, unmodified (no trimming).

    Returns:
        {'valid': bool, 'invalid': [{'symbol', 'index'}, ...]} with 1-based
        positions. A space is reported as symbol ' ' and the interface renders
        that as "space".
    """
    invalid = [
        {"symbol": char, "index": i}
        for i, char in enumerate(input_str, start=1)
        if char not in DFA["alphabet"]
    ]
    return {"valid": not invalid, "invalid": invalid}


def simulate(input_str: str) -> Dict[str, Any]:
    """
    Simulate the machine over a valid input string, one symbol at a time.

    Args:
        input_str: a string containing only '0' and '1'.

    Returns:
        {'input', 'startState', 'steps', 'flow', 'finalState', 'accepted',
         'description'}.
        steps is [{'step', 'from', 'symbol', 'to'}] numbered from 1; flow has
        one more entry than steps, starting at the start state.
        Empty input yields steps [], flow ['q0'], finalState 'q0', and
        accepted False, because the empty string ends in nothing and is not a
        member of the language.
    """
    start_state = DFA["start"]

    def result(steps: List[Dict[str, Any]], flow: List[str], final_state: str) -> Dict[str, Any]:
        return {
            "input": input_str,
            "startState": start_state,
            "steps": steps,
            "flow": flow,
            "finalState": final_state,
            "accepted": final_state in DFA["accepting"],
            "description": ACCEPTED_DESCRIPTION,
        }

    # Empty input: no transitions at all, so the machine is still at the start.
    if not input_str:
        return result([], [start_state], start_state)

    steps: List[Dict[str, Any]] = []
    flow = [start_state]
    current = start_state

    for i, symbol in enumerate(input_str, start=1):
        nxt = DFA["delta"][current][symbol]
        steps.append({"step": i, "from": current, "symbol": symbol, "to": nxt})
        current = nxt
        flow.append(current)

    return result(steps, flow, current)


def check(input_str: str) -> Dict[str, Any]:
    """
    Validate, then simulate. This is what the bridge calls.

    Args:
        input_str: the string to check, unmodified.

    Returns:
        {'status': 'invalid', 'errors': [{'symbol', 'index'}, ...]} when any
        symbol is outside the alphabet - an invalid string is never reported as
        accepted or rejected, because the machine never saw it.
        Otherwise {'status': 'valid', 'input', 'start', 'steps', 'flow',
        'final', 'accepted', 'description'}.
    """
    validation = validate(input_str)
    if not validation["valid"]:
        return {"status": "invalid", "errors": validation["invalid"]}

    sim = simulate(input_str)
    return {
        "status": "valid",
        "input": sim["input"],
        "start": sim["startState"],
        "steps": sim["steps"],
        "flow": sim["flow"],
        "final": sim["finalState"],
        "accepted": sim["accepted"],
        "description": sim["description"],
    }


# The oracle. Every case states the expected acceptance AND the expected final
# state, because accepting with the wrong final state would mean the transition
# function and the acceptance rule had drifted apart.
TEST_CASES = [
    # The ten cases named in the assignment brief.
    {"input": "1", "expected": "ACCEPTED", "final": "q1"},
    {"input": "0", "expected": "REJECTED", "final": "q0"},
    {"input": "11", "expected": "ACCEPTED", "final": "q1"},
    {"input": "10", "expected": "REJECTED", "final": "q0"},
    {"input": "101", "expected": "ACCEPTED", "final": "q1"},
    {"input": "100", "expected": "REJECTED", "final": "q0"},
    {"input": "111", "expected": "ACCEPTED", "final": "q1"},
    {"input": "110", "expected": "REJECTED", "final": "q0"},
    {"input": "10101", "expected": "ACCEPTED", "final": "q1"},
    {"input": "10100", "expected": "REJECTED", "final": "q0"},
    # The cases that show "ends in 1" is not "is exactly 101".
    {"input": "0101", "expected": "ACCEPTED", "final": "q1"},
    {"input": "1010", "expected": "REJECTED", "final": "q0"},
    {"input": "0001", "expected": "ACCEPTED", "final": "q1"},
    {"input": "11000", "expected": "REJECTED", "final": "q0"},
    {"input": "11111", "expected": "ACCEPTED", "final": "q1"},
    {"input": "00000", "expected": "REJECTED", "final": "q0"},
    # Single symbols, and longer runs of each.
    {"input": "0111111", "expected": "ACCEPTED", "final": "q1"},
    {"input": "0000000000000", "expected": "REJECTED", "final": "q0"},
    {"input": "101101", "expected": "ACCEPTED", "final": "q1"},
    {"input": "1101011", "expected": "ACCEPTED", "final": "q1"},
    # Empty input: no symbol read, still at the start, not accepted.
    {"input": "", "expected": "REJECTED", "final": "q0"},
    # Validation runs before the automaton, so none of these reach it.
    {"input": "1021", "expected": "INVALID", "final": None},
    {"input": "10a1", "expected": "INVALID", "final": None},
    {"input": "10 1", "expected": "INVALID", "final": None},
    {"input": "1 1", "expected": "INVALID", "final": None},
    {"input": "101 ", "expected": "INVALID", "final": None},
    {"input": "2", "expected": "INVALID", "final": None},
    {"input": "abc", "expected": "INVALID", "final": None},
    {"input": " ", "expected": "INVALID", "final": None},
]


def run_tests() -> Dict[str, Any]:
    """
    Run every case in TEST_CASES through check().

    Returns:
        {'total', 'passed', 'results'} where each result is
        {'input', 'expected', 'actual', 'final', 'finalOk', 'pass'}. `actual`
        always comes from check(), never from looking at the last character.
    """
    results = []
    passed = 0

    for case in TEST_CASES:
        value = case["input"]
        expected = case["expected"]
        result = check(value)

        if result["status"] == "invalid":
            actual = "INVALID"
            final_ok = True
        else:
            actual = "ACCEPTED" if result["accepted"] else "REJECTED"
            # The final state is part of the expectation, not just accept/reject.
            final_ok = case["final"] is None or result["final"] == case["final"]

        ok = actual == expected and final_ok
        if ok:
            passed += 1

        results.append({
            "input": value,
            "expected": expected,
            "actual": actual,
            "final": None if result["status"] == "invalid" else result["final"],
            "finalOk": final_ok,
            "pass": ok,
        })

    return {"total": len(TEST_CASES), "passed": passed, "results": results}


if __name__ == "__main__":
    report = run_tests()
    for number, row in enumerate(report["results"], start=1):
        label = row["input"] if row["input"] else "(empty)"
        final = row["final"] or "-"
        mark = "PASS" if row["pass"] else "FAIL"
        print(f"{number:>2}  {label!r:<18} expected {row['expected']:<9}"
              f" actual {row['actual']:<9} final {final:<4} {mark}")
    print(f"\n{report['passed']} of {report['total']} passed.")
